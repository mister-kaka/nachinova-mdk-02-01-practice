def find_min(numbers):
    """Возвращает минимальное число в списке. Список не пустой."""
    if not numbers:
        return None
    minimum = numbers[0]
    for n in numbers:
        if n < minimum:
            minimum = n
    return minimum

def get_done_tasks():
    """Возвращает список названий задач со статусом 'done'."""
    conn = sqlite3.connect("app.db")
    conn.row_factory = sqlite3.Row
    try:
        cursor = conn.cursor()
        cursor.execute("SELECT title FROM tasks WHERE status = ?", ("done",))
        return [row["title"] for row in cursor.fetchall()]
    except sqlite3.Error as e:
        print(f"[DB ERROR] {e}")
        return []

    if request.method == 'POST':
        title = request.form.get('title', '').strip()
        if not title:
            return render_template('index.html', items=[], error='Пусто!')

        conn = sqlite3.connect('app.db')
        try:
            cursor = conn.cursor()
            cursor.execute('INSERT INTO tasks (title) VALUES (?)', (title,))
            conn.commit()
        except sqlite3.Error as e:
            conn.rollback()
            return render_template('index.html', items=[], error=str(e))
        finally:
            conn.close()

        return redirect(url_for('index'))

    return render_template('index.html', items=[])

    for x in lst:
        if type(x) == list:

                result.append(n)

from flask import Flask, render_template, request, redirect, url_for
import sqlite3

app = Flask(__name__)
DB_PATH = 'students.db'

@app.route('/students', methods=['GET', 'POST'])
def students():
    if request.method == 'POST':
        name = request.form.get('name', '').strip()
        grade_text = request.form.get('grade', '').strip()
        if name == '':
            return render_template('students.html', students=[], error='Имя не может быть пустым')
        try:
            grade = int(grade_text)
        except ValueError:
            return render_template('students.html', students=[], error='Оценка должна быть целым числом от 1 до 5')
        if grade < 1 or grade > 5:
            return render_template('students.html', students=[], error='Оценка должна быть целым числом от 1 до 5')
        conn = sqlite3.connect(DB_PATH)
        try:
            cursor = conn.cursor()
            cursor.execute('INSERT INTO students (name, grade) VALUES (?, ?)', (name, grade))
            conn.commit()
        except sqlite3.Error as e:
            conn.rollback()
            return render_template('students.html', students=[], error=str(e))
        finally:
            conn.close()
        return redirect(url_for('students'))

    conn = sqlite3.connect(DB_PATH)
    try:
        cursor = conn.cursor()
        cursor.execute('SELECT id, name, grade FROM students')
        rows = cursor.fetchall()
    except sqlite3.Error as e:
        conn.rollback()
        return render_template('students.html', students=[], error=str(e))
    finally:
        conn.close()
    return render_template('students.html', students=rows)

def top_three_by_grade(students):
    top = []
    for name, grade in students:
        i = 0
        while i < len(top) and top[i][1] >= grade:
            i += 1
        if i < 3:
            top.insert(i, (name, grade))
            if len(top) > 3:
                top.pop()
    result = []
    for name, grade in top:
        result.append(name)
    return result

def count_chars(text):
    """Возвращает словарь: символ -> количество вхождений. Пробелы игнорируются."""
    result = {}
    for ch in text:
        if ch == " ":
            continue
        result[ch] = result.get(ch, 0) + 1
    return result

def invert_dict(d):
    """Меняет местами ключи и значения. Если значений несколько - собирает список ключей."""
    result = {}
    for key, value in d.items():
        if value not in result:
            result[value] = []
        result[value].append(key)
    return result

def update_price(product_id, new_price):
    """Обновляет цену. Возвращает rowcount. При ошибке - rollback и 0."""
    conn = sqlite3.connect("shop.db")
    try:
        cursor = conn.cursor()
        cursor.execute(
            "UPDATE products SET price = ? WHERE id = ?",
            (new_price, product_id),
        )
        conn.commit()
        return cursor.rowcount
    except sqlite3.Error as e:
        conn.rollback()
        print(f"[DB ERROR] {e}")
        return 0
    finally:
        conn.close()

    conn = sqlite3.connect('shop.db')

        cursor = conn.cursor()
        cursor.execute('DELETE FROM items WHERE id = ?', (item_id,))

def stats(students):
    if len(students) == 0:
        return (None, None, None)
    min_grade = students[0][1]
    max_grade = students[0][1]
    total = 0
    for name, grade in students:
        if grade < min_grade:
            min_grade = grade
        if grade > max_grade:
            max_grade = grade
        total += grade
    avg = round(total / len(students), 2)
    return (min_grade, max_grade, avg)

def top_authors(books, n):
    counts = {}
    for title, author in books:
        if author in counts:
            counts[author] += 1
        else:
            counts[author] = 1
    pairs = []
    for author in counts:
        pairs.append([author, counts[author]])
    i = 0
    while i < len(pairs):
        j = 0
        while j < len(pairs) - 1:
            if pairs[j][1] < pairs[j + 1][1]:
                temp = pairs[j]
                pairs[j] = pairs[j + 1]
                pairs[j + 1] = temp
            elif pairs[j][1] == pairs[j + 1][1] and pairs[j][0] > pairs[j + 1][0]:
                temp = pairs[j]
                pairs[j] = pairs[j + 1]
                pairs[j + 1] = temp
            j += 1
        i += 1
    result = []
    i = 0
    while i < n and i < len(pairs):
        result.append(pairs[i][0])
        i += 1
    return result
